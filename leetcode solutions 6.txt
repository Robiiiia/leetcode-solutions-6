class Solution {
    public int[] twoSum(int[] nums, int target) {
        HashMap<Integer, Integer> map = new HashMap<>(); //we are creating hashmap
        for (int i = 0; i < nums.length; i++) { //stars a loop to iterate through evry element of the array
            int n = target - nums[i]; //the difference between target and the current element to find a needed number

            if (map.containsKey(n)) { //check if the complement n already exists in our hashmap
                return new int[]{map.get(n), i}; //if found, returns an array containing the previously stored index and the current index i
            }

            map.put(nums[i], i); //if not found, adds the current element and its index to the hashmapfor future lookups 
        }

        return new int[]{}; //returns an empty array if nosolutionis found
    }
}
